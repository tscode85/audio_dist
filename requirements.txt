# file: requirements.txt
# Pinned dependencies for the acoustic-gap toolkit.
# Versions are chosen to be mutually compatible and to work fully offline once
# model checkpoints are vendored (see setup/download_models.py).
#
# NOTE: install on a networked machine, then ship the environment + model_cache
# to the air-gapped host. Nothing below reaches the network at runtime.

# --- core numerics / audio I/O ---
numpy==1.26.4
scipy==1.11.4
pandas==2.1.4
soundfile==0.12.1
librosa==0.10.1

# --- deep-learning backbones ---
# The PyPI `fadtk` package does NOT implement PANN (any version). Real PANN
# (Kong et al. Cnn14) is provided by `frechet_audio_distance` as
# model_name="pann", which has loose torch pins — so we stay on the proven,
# lightweight torch 2.1.2 (cu121) stack. torchvision is required by
# frechet_audio_distance; 0.16.2 is the torch-2.1.2-compatible release.
torch==2.1.2
torchvision==0.16.2
torchaudio==2.1.2
transformers==4.36.2        # provides Wav2Vec2FeatureExtractor + WavLMForXVector

# --- Frechet Audio Distance / PANN / VGGish embeddings ---
# frechet_audio_distance is the FAD backend. model_name="pann" loads Cnn14
# (16 kHz variant); model_name="vggish" loads VGGish. It imports laion_clap and
# encodec at module load, so those (and torchlibrosa for the PANN model) are
# pinned explicitly. Python >=3.10 (the CUDA base image ships Python 3.10).
frechet_audio_distance==0.3.4
laion-clap==1.1.7
encodec==0.1.1
torchlibrosa==0.1.0

# --- optimal transport (multivariate Wasserstein / EMD) ---
POT==0.9.1

# --- reporting / plotting ---
matplotlib==3.8.2

# --- config ---
PyYAML==6.0.1

# --- testing ---
pytest==7.4.4
