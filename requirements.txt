# file: requirements.txt
# Pinned dependencies for the acoustic-gap toolkit.
# Versions are chosen to be mutually compatible and to work fully offline once
# model checkpoints are vendored (see setup/download_models.py).
#
# NOTE: install on a networked machine, then ship the environment + model_cache
# to the air-gapped host. Nothing below reaches the network at runtime.

# --- core numerics / audio I/O ---

scipy==1.11.4
pandas==2.1.4
soundfile==0.12.1
librosa==0.10.1

# --- deep-learning backbones ---
# fadtk 1.1.0 (which provides the PANN "panns-wavegram-logmel" model) requires
# torch>=2.3 + torchvision>=0.22; torchvision 0.22.0 pins torch==2.7.0, so the
# whole trio is aligned at 2.7.0. Install these from the CUDA 12.6 wheel index
# (see docker/Dockerfile) — CUDA 12.6 is forward-compatible with NVIDIA driver
# 580. numpy stays <2 (fadtk requirement).
torch==2.7.0
torchvision==0.22.0
torchaudio==2.7.0
transformers==4.49.0        # Wav2Vec2FeatureExtractor + WavLMForXVector; torch-2.7 compatible

# --- Frechet Audio Distance / PANN / VGGish embeddings ---
# fadtk is the primary FAD backend. 1.1.0 ships PANN ("panns-wavegram-logmel")
# and VGGish; Python <=3.13,>=3.10 (the CUDA base image ships Python 3.10).
fadtk==1.1.0
# frechet_audio_distance==0.3.4   # alternative backend (uncomment if preferred)

# --- optimal transport (multivariate Wasserstein / EMD) ---
POT==0.9.1

# --- reporting / plotting ---
matplotlib==3.8.2
